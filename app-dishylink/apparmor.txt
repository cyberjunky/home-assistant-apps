#include <tunables/global>

profile app-dishylink flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities. A custom profile grants none by default, so the ones the
  # start scripts need are listed: chown/fowner for `chown -R` on the data
  # directory, setuid/setgid for s6-setuidgid dropping to the dishylink user,
  # dac_override so root can traverse the user-owned recordings.
  capability chown,
  capability fowner,
  capability fsetid,
  capability setuid,
  capability setgid,
  capability dac_override,
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within your app
  /data/** rw,

  # Start new profiles for the two services
  /usr/bin/node cx -> dishylink,
  /usr/sbin/nginx cx -> nginx,

  # Dishylink browser host: web dashboard, history recorder and LAN/cloud proxies.
  profile dishylink flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    # Receive signals from S6-Overlay; tsx signals the esbuild helper it spawns.
    signal (receive) peer=*_app-dishylink,
    signal (send,receive) peer=*_app-dishylink//dishylink,

    # Dish and router on the LAN, starlink.com / celestrak.org / Cloudflare upstream,
    # and netlink for the interface listing (os.networkInterfaces).
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    /usr/bin/node mr,
    # Node's OpenSSL reads its config and CA bundle; ssl_certs covers only the
    # certificates. musl consults its library search path file if present.
    /etc/ssl/ r,
    /etc/ssl/** r,
    /etc/ld-musl-*.path r,
    /app/ r,
    /app/** r,
    /app/node_modules/@esbuild/*/bin/esbuild ix,

    # Recordings, pasted starlink.com session, self-device and router-address files.
    /data/historian/ rw,
    /data/historian/** rwk,

    /proc/ r,
    /proc/** r,
    /sys/devices/system/cpu/** r,
    /sys/fs/cgroup/** r,
    /dev/null rw,
    /dev/urandom r,
    /dev/tty rw,
    /etc/localtime r,
    /usr/share/zoneinfo/** r,
    /tmp/** rwk,
  }

  # nginx ingress proxy on 8099.
  profile nginx flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    # Receive signals from S6-Overlay; master <-> worker signalling.
    signal (receive) peer=*_app-dishylink,
    signal (send,receive) peer=*_app-dishylink//nginx,

    # The master runs as root, chowns its temp directories to the nginx user
    # and drops the workers to that user.
    capability chown,
    capability setuid,
    capability setgid,
    capability dac_override,

    network inet stream,
    network inet6 stream,

    /usr/sbin/nginx mr,
    /etc/nginx/ r,
    /etc/nginx/** r,
    /run/nginx/ rw,
    /run/nginx/** rwk,
    /var/lib/nginx/ rw,
    /var/lib/nginx/** rwk,
    /var/log/nginx/ rw,
    /var/log/nginx/** rw,
    /proc/ r,
    /proc/** r,
    /dev/null rw,
    /dev/urandom r,
    /dev/tty rw,
    /etc/localtime r,
    /tmp/** rwk,
  }
}
