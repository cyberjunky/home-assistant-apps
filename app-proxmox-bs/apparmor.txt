#include <tunables/global>

# Proxmox Backup Server runs as root and as the backup user, reads and writes its configuration,
# the datastores and its task logs, partitions and formats disks from its Disks page, and logs
# users in through PAM, so files, network and capabilities are broad. What this profile narrows is
# mounting: only datastore disks under /mnt/datastore; no other mount, and no changes to the kernel.
profile app-proxmox-bs flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  capability,
  file,
  network,
  signal,
  unix,
  ptrace (read, readby, trace, tracedby) peer=app-proxmox-bs,

  # Datastore disks, where PBS's Disks page puts them, and nothing else
  mount fstype=(ext4,xfs,btrfs) /dev/** -> /mnt/datastore/**,
  umount /mnt/datastore/**,

  # No loading of kernel modules, no kernel settings, no raw memory
  deny capability sys_module,
  deny /proc/sys/** w,
  deny /sys/** w,
  deny /dev/mem rw,
  deny /dev/kmem rw,
  deny /dev/port rw,
}
